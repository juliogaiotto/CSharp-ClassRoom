# 🗄️ SCRIPT SQL — InfraMind

``` SQL
-- Criar banco
CREATE DATABASE inframinddb CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE inframinddb;

-- =========================
-- TABELA: ROLES (PERFIS)
-- =========================
CREATE TABLE roles (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(50) NOT NULL UNIQUE
);

-- =========================
-- TABELA: USUARIOS
-- =========================
CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    senha_hash VARCHAR(255) NOT NULL,
    role_id INT NOT NULL,
    ativo BOOLEAN DEFAULT TRUE,
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (role_id) REFERENCES roles(id)
);

-- =========================
-- TABELA: CLIENTES (EMPRESAS)
-- =========================
CREATE TABLE clientes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    razao_social VARCHAR(150) NOT NULL,
    cnpj VARCHAR(20),
    email VARCHAR(100),
    telefone VARCHAR(20),
    endereco VARCHAR(200),
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- =========================
-- TABELA: FUNCIONARIOS
-- =========================
CREATE TABLE funcionarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(100),
    telefone VARCHAR(20),
    cliente_id INT NOT NULL,
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (cliente_id) REFERENCES clientes(id)
);

-- =========================
-- TABELA: EQUIPAMENTOS
-- =========================
CREATE TABLE equipamentos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    descricao VARCHAR(150) NOT NULL,
    serial_number VARCHAR(100),
    modelo VARCHAR(100),
    cliente_id INT NOT NULL,
    funcionario_id INT NULL,
    imagem VARCHAR(200),
    qr_code_token VARCHAR(200) UNIQUE,
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (cliente_id) REFERENCES clientes(id),
    FOREIGN KEY (funcionario_id) REFERENCES funcionarios(id)
);

-- =========================
-- TABELA: CHAMADOS
-- =========================
CREATE TABLE chamados (
    id INT AUTO_INCREMENT PRIMARY KEY,
    titulo VARCHAR(150) NOT NULL,
    descricao TEXT NOT NULL,
    status ENUM('Aberto', 'Em_Andamento', 'Fechado') DEFAULT 'Aberto',
    prioridade ENUM('Baixa', 'Media', 'Alta', 'Critica') DEFAULT 'Media',
    cliente_id INT NOT NULL,
    equipamento_id INT,
    tecnico_id INT,
    data_abertura DATETIME DEFAULT CURRENT_TIMESTAMP,
    data_fechamento DATETIME,
    FOREIGN KEY (cliente_id) REFERENCES clientes(id),
    FOREIGN KEY (equipamento_id) REFERENCES equipamentos(id),
    FOREIGN KEY (tecnico_id) REFERENCES usuarios(id)
);

-- =========================
-- TABELA: HISTORICO_CHAMADO
-- =========================
CREATE TABLE historico_chamado (
    id INT AUTO_INCREMENT PRIMARY KEY,
    chamado_id INT NOT NULL,
    status_anterior VARCHAR(50),
    status_novo VARCHAR(50),
    observacao TEXT,
    data_evento DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (chamado_id) REFERENCES chamados(id)
);

-- =========================
-- TABELA: SLA
-- =========================
CREATE TABLE sla (
    id INT AUTO_INCREMENT PRIMARY KEY,
    prioridade ENUM('Baixa', 'Media', 'Alta', 'Critica') NOT NULL,
    tempo_maximo_horas INT NOT NULL
);

-- =========================
-- TABELA: CONFIGURACAO_SLA_POR_CHAMADO
-- =========================
CREATE TABLE chamado_sla (
    id INT AUTO_INCREMENT PRIMARY KEY,
    chamado_id INT NOT NULL,
    sla_id INT NOT NULL,
    tempo_gasto_minutos INT,
    dentro_sla BOOLEAN,
    FOREIGN KEY (chamado_id) REFERENCES chamados(id),
    FOREIGN KEY (sla_id) REFERENCES sla(id)
);

-- =========================
-- TABELA: LOGS
-- =========================
CREATE TABLE logs (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nivel VARCHAR(20),
    mensagem TEXT,
    data_evento DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- =========================
-- DADOS INICIAIS
-- =========================
INSERT INTO roles (nome) VALUES ('Admin'), ('Tecnico');

INSERT INTO sla (prioridade, tempo_maximo_horas) VALUES
('Baixa', 72),
('Media', 48),
('Alta', 24),
('Critica', 4);
```

### 📐 RELACIONAMENTOS PRINCIPAIS

* usuarios → roles
* funcionarios → clientes
* equipamentos → clientes
* chamados → clientes
* chamados → equipamentos
* chamados → usuarios (tecnico)
* historico_chamado → chamados
* chamado_sla → chamados
* chamado_sla → sla
